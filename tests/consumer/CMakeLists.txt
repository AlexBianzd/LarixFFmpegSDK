cmake_minimum_required(VERSION 3.25)
project(LarixFFmpegSDKConsumer C)
find_package(LarixFFmpegSDK CONFIG REQUIRED)
add_executable(larix_ffmpeg_smoke smoke.c)
target_compile_features(larix_ffmpeg_smoke PRIVATE c_std_11)
target_link_libraries(larix_ffmpeg_smoke PRIVATE
  LarixFFmpegSDK::avutil LarixFFmpegSDK::avcodec LarixFFmpegSDK::avformat
  LarixFFmpegSDK::swresample LarixFFmpegSDK::swscale LarixFFmpegSDK::avfilter)
foreach(_component avutil avcodec avformat swresample swscale avfilter)
  get_target_property(_runtime LarixFFmpegSDK::${_component} IMPORTED_LOCATION)
  if(NOT EXISTS "${_runtime}")
    message(FATAL_ERROR "Imported Larix FFmpeg SDK component is incomplete: ${_component}")
  endif()
  if(WIN32)
    get_target_property(_implib LarixFFmpegSDK::${_component} IMPORTED_IMPLIB)
    if(NOT EXISTS "${_runtime}" OR NOT EXISTS "${_implib}")
      message(FATAL_ERROR "Imported Larix FFmpeg SDK component is incomplete: ${_component}")
    endif()
  endif()
endforeach()
add_executable(larix_ffmpeg_encoding encoding.c)
target_compile_features(larix_ffmpeg_encoding PRIVATE c_std_11)
target_link_libraries(larix_ffmpeg_encoding PRIVATE LarixFFmpegSDK::avcodec LarixFFmpegSDK::avformat LarixFFmpegSDK::avutil)
add_executable(larix_ffmpeg_hdr hdr.c)
target_compile_features(larix_ffmpeg_hdr PRIVATE c_std_11)
target_link_libraries(larix_ffmpeg_hdr PRIVATE LarixFFmpegSDK::avfilter LarixFFmpegSDK::avutil)
if(UNIX)
  target_link_libraries(larix_ffmpeg_hdr PRIVATE m)
endif()
# Negative fixture: the SDK must never dynamically discover this system encoder.
add_library(larix_foreign_x265 SHARED ../fixtures/foreign_x265.c)
set_target_properties(larix_foreign_x265 PROPERTIES PREFIX "lib" OUTPUT_NAME "x265_main12")
