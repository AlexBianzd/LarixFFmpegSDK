cmake_minimum_required(VERSION 3.25)
project(LarixFFmpegSDKConsumer LANGUAGES C CXX)
enable_testing()
find_package(LarixFFmpegSDK CONFIG REQUIRED)
add_executable(larix_ffmpeg_smoke smoke.c)
target_compile_features(larix_ffmpeg_smoke PRIVATE c_std_11)
target_link_libraries(larix_ffmpeg_smoke PRIVATE
  LarixFFmpegSDK::avutil LarixFFmpegSDK::avcodec LarixFFmpegSDK::avformat
  LarixFFmpegSDK::swresample LarixFFmpegSDK::swscale)
add_executable(presentation_transport presentation.c)
target_compile_features(presentation_transport PRIVATE c_std_11)
target_link_libraries(presentation_transport PRIVATE LarixFFmpegSDK::avutil)
add_test(NAME presentation_transport COMMAND presentation_transport)
add_executable(presentation_cpp presentation_cpp.cpp)
target_compile_features(presentation_cpp PRIVATE cxx_std_17)
target_link_libraries(presentation_cpp PRIVATE LarixFFmpegSDK::avutil)
add_test(NAME presentation_cpp COMMAND presentation_cpp)
option(LARIX_PRESENTATION_FORCE_FAILURE "Force the Release REQUIRE witness to fail" OFF)
if(LARIX_PRESENTATION_FORCE_FAILURE)
  target_compile_definitions(presentation_transport PRIVATE LARIX_PRESENTATION_FORCE_FAILURE=1)
  target_compile_definitions(presentation_cpp PRIVATE LARIX_PRESENTATION_FORCE_FAILURE=1)
endif()
foreach(_component avutil avcodec avformat swresample swscale)
  get_target_property(_runtime LarixFFmpegSDK::${_component} IMPORTED_LOCATION)
  if(NOT EXISTS "${_runtime}")
    message(FATAL_ERROR "Imported Larix FFmpeg SDK component is incomplete: ${_component}")
  endif()
  if(WIN32)
    get_target_property(_implib LarixFFmpegSDK::${_component} IMPORTED_IMPLIB)
    if(NOT EXISTS "${_runtime}" OR NOT EXISTS "${_implib}")
      message(FATAL_ERROR "Imported Larix FFmpeg SDK component is incomplete: ${_component}")
    endif()
  endif()
endforeach()
