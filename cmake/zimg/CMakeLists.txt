cmake_minimum_required(VERSION 3.25)
project(LarixZimgPackaging LANGUAGES CXX)
if(NOT IS_DIRECTORY "${ZIMG_SOURCE}/src/zimg")
  message(FATAL_ERROR "A verified ZIMG_SOURCE is required")
endif()
# Build the upstream scalar implementation without editing the source tree.
# This deliberately avoids a build-host-specific ISA baseline.
file(GLOB_RECURSE _sources CONFIGURE_DEPENDS "${ZIMG_SOURCE}/src/zimg/*.cpp")
list(FILTER _sources EXCLUDE REGEX "/(x86|arm)/")
add_library(zimg STATIC ${_sources})
target_compile_features(zimg PRIVATE cxx_std_17)
target_include_directories(zimg PRIVATE "${ZIMG_SOURCE}/src/zimg" "${ZIMG_SOURCE}/src/zimg/api")
set_target_properties(zimg PROPERTIES POSITION_INDEPENDENT_CODE ON)
if(MSVC)
  target_compile_options(zimg PRIVATE /O2 /fp:strict /experimental:deterministic /Brepro "/pathmap:${ZIMG_SOURCE}=zimg-source")
  target_compile_definitions(zimg PRIVATE _CRT_SECURE_NO_WARNINGS _SCL_SECURE_NO_WARNINGS)
else()
  target_compile_options(zimg PRIVATE -O2 -fvisibility=hidden "-ffile-prefix-map=${ZIMG_SOURCE}=zimg-source")
endif()
install(TARGETS zimg ARCHIVE DESTINATION lib)
install(FILES "${ZIMG_SOURCE}/src/zimg/api/zimg.h" DESTINATION include)
